 * Background task using javafx.concurrent.Task to compute and generate 
 * a paying customer's billing history asynchronously without blocking the JavaFX UI thread.
 * 
 * @author Yun Shiang
 * @version 2.0
 */

import javafx.concurrent.Task;

public class BillingHistoryTask extends Task<String> {
    private PayingCustomer payingCustomer;
    private double magazineBaseCost;

    public BillingHistoryTask(PayingCustomer payingCustomer, double magazineBaseCost) {
        this.payingCustomer = payingCustomer;
        this.magazineBaseCost = magazineBaseCost;
    }

    @Override
    protected String call() throws Exception {
        // Simulate background computation progress
        updateMessage("Computing billing history...");
        for (int i = 1; i <= 3; i++) {
            Thread.sleep(200); // Simulate background processing delay
            updateProgress(i, 3);
        }

        if (payingCustomer == null) {
            return "Error: No paying customer selected for billing history.";
        }

        // Build the itemized billing history statement
        StringBuilder sb = new StringBuilder();
        sb.append("=== ASYNCHRONOUS BILLING HISTORY REPORT ===\n");
        sb.append("Customer Name: ").append(payingCustomer.getName()).append("\n");
        sb.append("Email: ").append(payingCustomer.getEmail()).append("\n");
        sb.append("Payment Method: ").append(payingCustomer.getPaymentMethod().getPaymentDetails()).append("\n\n");
        
        sb.append("Itemized Charges (4-Week Billing Cycle):\n");
        sb.append(" - Main Magazine Base Fee: $").append(String.format("%.2f", magazineBaseCost * 4)).append("\n");
        sb.append(" - Primary Supplements Fee: $").append(String.format("%.2f", payingCustomer.getWeeklySupplementsCost() * 4)).append("\n");

        if (!payingCustomer.getAssociateCustomers().isEmpty()) {
            sb.append("\nSponsored Associate Accounts:\n");
            for (AssociateCustomer assoc : payingCustomer.getAssociateCustomers()) {
                sb.append(" * ").append(assoc.getName()).append(" - Supplements: $")
                  .append(String.format("%.2f", assoc.getWeeklySupplementsCost() * 4)).append("\n");
            }
        }

        double totalMonthlyCost = payingCustomer.calculateMonthlyTotal(magazineBaseCost);
        sb.append("\nTotal Monthly Billing Amount: $").append(String.format("%.2f", totalMonthlyCost)).append("\n");
        sb.append("===========================================\n");

        return sb.toString();
    }
}

import javafx.stage.FileChooser;
import javafx.stage.Stage;

import java.io.*;

public class FileHandler {

    /**
     * Prompts the user with a FileChooser to save the current Magazine object to disk.
     * 
     * @param ownerStage the parent JavaFX stage for the dialog window
     * @param magazine   the Magazine instance to serialize and save
     */
    public static void saveMagazine(Stage ownerStage, Magazine magazine) {
        FileChooser fileChooser = new FileChooser();
        fileChooser.setTitle("Save Magazine Service Data");
        fileChooser.getExtensionFilters().add(
            new FileChooser.ExtensionFilter("Serialized Data Files (*.ser)", "*.ser")
        );
        fileChooser.setInitialFileName("magazine_data.ser");

        File file = fileChooser.showSaveDialog(ownerStage);
        if (file != null) {
            try (ObjectOutputStream oos = new ObjectOutputStream(new FileOutputStream(file))) {
                oos.writeObject(magazine);
                System.out.println("[SUCCESS] Magazine data successfully saved to " + file.getAbsolutePath());
            } catch (IOException e) {
                System.err.println("[ERROR] Failed to save file: " + e.getMessage());
                e.printStackTrace();
            }
        }
    }

    /**
     * Prompts the user with a FileChooser to load a saved Magazine object from disk.
     * 
     * @param ownerStage the parent JavaFX stage for the dialog window
     * @return the deserialized Magazine instance, or null if loading was cancelled or failed
     */
    public static Magazine loadMagazine(Stage ownerStage) {
        FileChooser fileChooser = new FileChooser();
        fileChooser.setTitle("Open Magazine Service Data");
        fileChooser.getExtensionFilters().add(
            new FileChooser.ExtensionFilter("Serialized Data Files (*.ser)", "*.ser")
        );

        File file = fileChooser.showOpenDialog(ownerStage);
        if (file != null) {
            try (ObjectInputStream ois = new ObjectInputStream(new FileInputStream(file))) {
                Magazine magazine = (Magazine) ois.readObject();
                System.out.println("[SUCCESS] Magazine data successfully loaded from " + file.getAbsolutePath());
                return magazine;
            } catch (IOException | ClassNotFoundException e) {
                System.err.println("[ERROR] Failed to load file: " + e.getMessage());
                e.printStackTrace();
            }
        }
        return null;
    }
}

 * Core domain class managing magazine pricing, available supplements, and customer base.
 * File Name: Magazine.java
 */

import java.io.Serializable;
import java.util.ArrayList;
import java.util.List;


public class Magazine implements Serializable {
    private static final long serialVersionUID = 1L;

    /** The base weekly subscription cost for the main magazine. */
    private double weeklyBaseCost;

    /** The list of all supplements available for subscription. */
    private List<Supplement> availableSupplements;

    /** The list of all registered active customers (both paying and associate). */
    private List<Customer> customers;

    /**
     * Constructs a new Magazine service with a specified weekly base cost.
     *
     * @param weeklyBaseCost the weekly price for the main magazine issue
     */
    public Magazine(double weeklyBaseCost) {
        this.weeklyBaseCost = weeklyBaseCost;
        this.availableSupplements = new ArrayList<>();
        this.customers = new ArrayList<>();
    }

    /**
     * Retrieves the base weekly subscription cost of the main magazine.
     *
     * @return the weekly base cost as a double
     */
    public double getWeeklyBaseCost() { return weeklyBaseCost; }

    /**
     * Updates the base weekly subscription cost of the main magazine.
     *
     * @param weeklyBaseCost the new weekly base cost to set
     */
    public void setWeeklyBaseCost(double weeklyBaseCost) { this.weeklyBaseCost = weeklyBaseCost; }

    /**
     * Retrieves the list of all available supplements offered by the magazine.
     *
     * @return a List of Supplement objects
     */
    public List<Supplement> getAvailableSupplements() { return availableSupplements; }

    /**
     * Retrieves the list of all active registered customers.
     *
     * @return a List of Customer objects
     */
    public List<Customer> getCustomers() { return customers; }

    /**
     * Adds a new supplement to the magazine's catalog if it is not already present.
     *
     * @param supplement the Supplement instance to add
     * @return true if the supplement was added successfully; false if it already exists
     */
    public boolean addSupplement(Supplement supplement) {
        if (!availableSupplements.contains(supplement)) {
            return availableSupplements.add(supplement);
        }
        return false;
    }

    /**
     * Removes an existing supplement from the magazine's catalog and cascades the removal 
     * from any customer's subscribed/interested supplement list.
     *
     * @param supplement the Supplement instance to remove
     * @return true if the supplement was successfully removed; false if null or not found
     */
    public boolean removeSupplement(Supplement supplement) {
        if (supplement == null || !availableSupplements.contains(supplement)) {
            return false;
        }

        // Cascade removal: remove this supplement from every customer's subscription list
        for (Customer customer : customers) {
            if (customer.getInterestedSupplements() != null) {
                customer.removeSupplement(supplement);
            }
        }

        // Finally, remove the supplement from the magazine's global catalog
        return availableSupplements.remove(supplement);
    }

    /**
     * Registers a new customer with the magazine service if not already registered.
     *
     * @param customer the Customer instance to add
     * @return true if the customer was registered successfully; false if already registered
     */
    public boolean addCustomer(Customer customer) {
        if (!customers.contains(customer)) {
            return customers.add(customer);
        }
        return false;
    }

    /**
     * Removes an existing customer from the magazine service.
     * <p>
     * If the customer is a PayingCustomer, a cascade deletion occurs to remove all 
     * associated customers paid for by them. If the customer is an AssociateCustomer, 
     * they are unlinked from their PayingCustomer before removal.
     * </p>
     *
     * @param customer the Customer instance to remove
     * @return true if the customer was successfully removed; false if null or not found
     */
    public boolean removeCustomer(Customer customer) {
        if (customer == null || !customers.contains(customer)) {
            return false;
        }

        if (customer instanceof PayingCustomer) {
            PayingCustomer pc = (PayingCustomer) customer;
            
            // Cascade delete: remove all associated customers paid for by this customer
            List<AssociateCustomer> associates = new ArrayList<>(pc.getAssociateCustomers());
            for (AssociateCustomer assoc : associates) {
                pc.removeAssociateCustomer(assoc);
                customers.remove(assoc); // Remove associate from active magazine customer list
            }
        } else if (customer instanceof AssociateCustomer) {
            AssociateCustomer ac = (AssociateCustomer) customer;
            if (ac.getPayingCustomer() != null) {
                ac.getPayingCustomer().removeAssociateCustomer(ac);
            }
        }

        return customers.remove(customer);
    }
}

import java.io.Serializable;
import java.util.ArrayList;
import java.util.List;

public class PayingCustomer extends Customer implements Serializable {
    private static final long serialVersionUID = 1L;

    /** The payment method registered to this customer for automatic billing. */
    private PaymentMethod paymentMethod;

    /** List of associate customers sponsored by this paying customer. */
    private List<AssociateCustomer> associateCustomers;

    /**
     * Constructs a PayingCustomer with specified name, email address, and payment method.
     *
     * @param name          the full name of the paying customer
     * @param email         the email address of the paying customer
     * @param paymentMethod the payment method used to process invoices
     */
    public PayingCustomer(String name, String email, Address address, PaymentMethod paymentMethod) {
        super(name, email, address);
        this.paymentMethod = paymentMethod;
        this.associateCustomers = new ArrayList<>();
    }

    /**
     * Gets the current payment method linked to this account.
     *
     * @return the associated {@link PaymentMethod}
     */
    public PaymentMethod getPaymentMethod() { return paymentMethod; }

    /**
     * Updates the payment method for this account.
     *
     * @param paymentMethod the new {@link PaymentMethod} to set
     */
    public void setPaymentMethod(PaymentMethod paymentMethod) { this.paymentMethod = paymentMethod; }

    /**
     * Retrieves the list of associate customers sponsored by this paying customer.
     *
     * @return a list of {@link AssociateCustomer} instances
     */
    public List<AssociateCustomer> getAssociateCustomers() { return associateCustomers; }

    /**
     * Links an associate customer to this paying customer's account.
     * <p>
     * Ensures bidirectional linkage by updating the associate's paying customer reference.
     * </p>
     *
     * @param associate the {@link AssociateCustomer} to be added
     */
    public void addAssociateCustomer(AssociateCustomer associate) {
        if (!associateCustomers.contains(associate)) {
            associateCustomers.add(associate);
            associate.setPayingCustomer(this);
        }
    }

    /**
     * Unlinks an associate customer from this paying customer's account.
     * <p>
     * Ensures bidirectional unlinking by clearing the associate's paying customer reference.
     * </p>
     *
     * @param associate the {@link AssociateCustomer} to be removed
     */
    public void removeAssociateCustomer(AssociateCustomer associate) {
        if (associateCustomers.remove(associate)) {
            associate.setPayingCustomer(null);
        }
    }

    /**
     * Calculates the total monthly charge for this primary customer and all linked associates.
     * <p>
     * The total is computed by summing the weekly base magazine cost and supplement costs 
     * for the primary customer and each associate, multiplied by 4 weeks.
     * </p>
     *
     * @param magazineBaseCost the weekly base cost of the main magazine subscription
     * @return the total calculated monthly cost ($) across 4 weeks
     */
    public double calculateMonthlyTotal(double magazineBaseCost) {
        double weeklyTotal = magazineBaseCost + getWeeklySupplementsCost();
        for (AssociateCustomer assoc : associateCustomers) {
            weeklyTotal += magazineBaseCost + assoc.getWeeklySupplementsCost();
        }
        return weeklyTotal * 4;
    }

    /**
     * Generates a fully itemized monthly billing statement email formatted for console output.
     *
     * @param magazineBaseCost the weekly base cost of the main magazine subscription
     * @return a formatted email String containing itemized charges for primary and associate accounts
     */
    public String generateMonthlyBillingEmail(double magazineBaseCost) {
        StringBuilder sb = new StringBuilder();
        sb.append("To: ").append(getEmail()).append("\n");
        sb.append("Subject: Monthly Subscription Statement & Invoice\n");
        sb.append("Dear ").append(getName()).append(",\n\n");
        sb.append("Billing Address: ").append(getAddress().toString()).append("\n");
        sb.append("Below is your itemized monthly billing statement (4-Week Period):\n\n");

        // Primary account holder itemization
        sb.append("Primary Customer: ").append(getName()).append("\n");
        sb.append(" - Main Magazine Base Fee (4 weeks): $")
          .append(String.format("%.2f", magazineBaseCost * 4)).append("\n");
        sb.append(" - Supplements Fee (4 weeks): $")
          .append(String.format("%.2f", getWeeklySupplementsCost() * 4)).append("\n");

        // Associate customers itemization
        if (!associateCustomers.isEmpty()) {
            sb.append("\nAssociate Accounts Covered:\n");
            for (AssociateCustomer assoc : associateCustomers) {
                sb.append(" * Associate: ").append(assoc.getName()).append("(").append(assoc.getEmail()).append(")\n");
                sb.append("   - Delivery Address: ").append(assoc.getAddress().toString()).append("\n");
                sb.append("   - Main Magazine Base Fee (4 weeks): $")
                  .append(String.format("%.2f", magazineBaseCost * 4)).append("\n");
                sb.append("   - Supplements Fee (4 weeks): $")
                  .append(String.format("%.2f", assoc.getWeeklySupplementsCost() * 4)).append("\n");
            }
        }

        double totalMonthlyCost = calculateMonthlyTotal(magazineBaseCost);
        sb.append("\nTotal Monthly Charge: $").append(String.format("%.2f", totalMonthlyCost)).append("\n");
        sb.append("Payment Method Charged: ").append(paymentMethod.getPaymentDetails()).append("\n");
        sb.append("--------------------------------------------------\n");
        return sb.toString();
    }
}
